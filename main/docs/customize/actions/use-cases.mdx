## Management API with Token Caching

A common pattern is to call the Auth0 Management API from within an Action to fetch roles, update user metadata, or retrieve app-specific data. Cache the token using `api.cache` to avoid fetching a new one on every execution.

```js Post-login: call Management API with cached token
exports.onExecutePostLogin = async (event, api) => {
  const ManagementClient = require('auth0').ManagementClient;

  // Try to get a cached token first
  const cached = api.cache.get('mgmt_token');
  let token = cached?.value;

  if (!token) {
    // Fetch a new token using client credentials stored as Action secrets
    const response = await fetch(`https://${event.secrets.AUTH0_DOMAIN}/oauth/token`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        grant_type: 'client_credentials',
        client_id: event.secrets.MGMT_CLIENT_ID,
        client_secret: event.secrets.MGMT_CLIENT_SECRET,
        audience: `https://${event.secrets.AUTH0_DOMAIN}/api/v2/`,
      }),
    });
    const data = await response.json();
    token = data.access_token;

    // Cache token for 23 hours (tokens expire in 24h by default)
    api.cache.set('mgmt_token', token, { ttl: 82800000 });
  }

  // Use the token to call the Management API
  const management = new ManagementClient({
    token,
    domain: event.secrets.AUTH0_DOMAIN,
  });

  const roles = await management.users.getRoles({ id: event.user.user_id });
  if (roles.data.some(r => r.name === 'admin')) {
    api.accessToken.setCustomClaim('https://example.com/is_admin', true);
  }
};
```

**Required Action secrets:** `AUTH0_DOMAIN`, `MGMT_CLIENT_ID`, `MGMT_CLIENT_SECRET`

**Required npm module:** `auth0` — add it under Action dependencies before deploying.


## Access Control

Allow access only on weekdays

```js Post-login
exports.onExecutePostLogin = async (event, api) => {
  const day = new Date().getDay(); // 0 = Sunday, 6 = Saturday
  if (day === 0 || day === 6) {
    api.access.deny('Access is only allowed on weekdays.');
  }
};
```

Allow access only from inside the corporate network

```js Post-login
exports.onExecutePostLogin = async (event, api) => {
  const CORPORATE_NETWORK = '192.168.1.';
  if (!event.request.ip.startsWith(CORPORATE_NETWORK)) {
    api.access.deny('Access is restricted to the corporate network.');
  }
};
```

Allow only specific email domains to log in

```js Post-login
exports.onExecutePostLogin = async (event, api) => {
  const allowedDomain = 'yourcompany.com';
  if (!event.user.email?.endsWith(`@${allowedDomain}`)) {
    api.access.deny(`Access is restricted to ${allowedDomain} accounts.`);
  }
};
```

## API Authorization

Add user roles to the Access Token

```js Post-login
exports.onExecutePostLogin = async (event, api) => {
  const roles = event.authorization?.roles ?? [];
  api.accessToken.setCustomClaim('https://example.com/roles', roles);
};
```

Add user email to the Access Token

```js Post-login
exports.onExecutePostLogin = async (event, api) => {
  api.accessToken.setCustomClaim('https://example.com/email', event.user.email);
};
```

Deny access to anyone calling a specific API

```js Post-login
exports.onExecutePostLogin = async (event, api) => {
  const restrictedAudience = 'https://restricted-api.example.com';
  if (event.resource_server?.identifier === restrictedAudience) {
    api.access.deny('You do not have permission to access this API.');
  }
};
```

Add custom claims to a token

```js Post-login
exports.onExecutePostLogin = async (event, api) => {
  api.accessToken.setCustomClaim('https://example.com/plan', event.user.app_metadata?.plan ?? 'free');
  api.idToken.setCustomClaim('https://example.com/preferred_language', event.user.user_metadata?.language ?? 'en');
};
```

Exchange an existing Auth0 token for another audience

```js Custom-token-exchange
exports.onExecuteCustomTokenExchange = async (event, api) => {
  // Validate the incoming subject token against your own system
  const response = await fetch('https://api.example.com/validate-token', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ token: event.transaction?.subject_token }),
  });

  if (!response.ok) {
    api.access.deny('Subject token is invalid or expired.');
    return;
  }

  const { userId, audience } = await response.json();
  // Add claims to the issued Auth0 token
  api.accessToken.setCustomClaim('https://example.com/original_audience', audience);
  api.accessToken.setCustomClaim('https://example.com/user_id', userId);
};
```

## Compliance

Redirect users to a consent page before login completes

```js Post-login
exports.onExecutePostLogin = async (event, api) => {
  const hasConsented = event.user.app_metadata?.gdpr_consent === true;
  if (!hasConsented) {
    const token = api.redirect.encodeToken({
      secret: event.secrets.REDIRECT_SECRET,
      payload: { userId: event.user.user_id },
      expiresInSeconds: 300,
    });
    api.redirect.sendUserTo('https://consent.example.com', { query: { token } });
  }
};

exports.onContinuePostLogin = async (event, api) => {
  // After user returns from consent page
  await api.user.setAppMetadata('gdpr_consent', true);
};
```

Require step-up authentication for sensitive transactions

```js Post-login
exports.onExecutePostLogin = async (event, api) => {
  const requestedAcr = event.transaction?.acr_values ?? [];
  const requiresMfa = requestedAcr.includes('http://schemas.openid.net/pape/policies/2007/06/multi-factor');
  if (requiresMfa) {
    api.authentication.challengeWith({ type: 'otp' });
  }
};
```

## Multifactor

Require MFA for users outside the corporate network

```js Post-login
exports.onExecutePostLogin = async (event, api) => {
  const CORPORATE_NETWORK = '192.168.1.';
  if (!event.request.ip.startsWith(CORPORATE_NETWORK)) {
    api.multifactor.enable('any', { allowRememberBrowser: false });
  }
};
```

Send a verification email after password change

```js Post-change-password
exports.onExecutePostChangePassword = async (event) => {
  const nodemailer = require('nodemailer');
  const transporter = nodemailer.createTransport({
    host: event.secrets.SMTP_HOST,
    port: 587,
    auth: { user: event.secrets.SMTP_USER, pass: event.secrets.SMTP_PASS },
  });
  await transporter.sendMail({
    from: 'no-reply@example.com',
    to: event.user.email,
    subject: 'Your password was changed',
    text: 'Your Auth0 account password was recently changed. If this was not you, contact support immediately.',
  });
};
```

## User Profile and Identity

Set a default profile picture for all users

```js Post-login
exports.onExecutePostLogin = async (event, api) => {
  if (!event.user.picture) {
    api.user.setUserMetadata('picture', 'https://example.com/assets/default-avatar.png');
  }
};
```

Retrieve user preferences from an external source at registration

```js Pre-user-registration
exports.onExecutePreUserRegistration = async (event, api) => {
  const response = await fetch(`https://api.example.com/users?email=${event.user.email}`);
  if (response.ok) {
    const data = await response.json();
    api.user.setUserMetadata('external_id', data.id);
    api.user.setAppMetadata('plan', data.plan);
  }
};
```

## Advanced Integrations

Migrate users from a legacy system into Auth0

```js Post-login
exports.onExecutePostLogin = async (event, api) => {
  if (event.user.app_metadata?.migrated) return;

  const response = await fetch('https://legacy-api.example.com/authenticate', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ email: event.user.email }),
  });

  if (response.ok) {
    await api.user.setAppMetadata('migrated', true);
    await api.user.setAppMetadata('legacy_id', (await response.json()).id);
  }
};
```

Re-use an external identity provider via Custom Token Exchange

```js Custom-token-exchange
exports.onExecuteCustomTokenExchange = async (event, api) => {
  const subjectToken = event.transaction?.subject_token;

  const response = await fetch('https://external-idp.example.com/verify', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${subjectToken}` },
  });

  if (!response.ok) {
    api.access.deny('Subject token could not be verified.');
    return;
  }

  const identity = await response.json();
  api.accessToken.setCustomClaim('https://example.com/external_id', identity.id);
};
```
